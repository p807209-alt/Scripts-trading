local function moveBypass(target, speed)
    local h, hum = getHRP(), getHum()
    if not h or not hum or hum.Health <= 0 then return 100 end
    if hum.WalkSpeed <= 0.1 then 
        h.AssemblyLinearVelocity = Vector3.new(0, h.AssemblyLinearVelocity.Y, 0)
        return (target - h.Position).Magnitude
    end
    hum.WalkSpeed = speed
    local dir = (target - h.Position)
    local dist = Vector3.new(dir.X, 0, dir.Z)
    if dist.Magnitude > 0.5 then
        h.AssemblyLinearVelocity = Vector3.new(dist.Unit.X * speed, h.AssemblyLinearVelocity.Y, dist.Unit.Z * speed)
        hum:MoveTo(target)
    end
    return dist.Magnitude
end

        local target = isLeft and POS_L[phase] or POS_R[phase]
        if not target then return end
        local fwd = (Config.CurrentMode == "Normal" and Config.NormalFwd or Config.DesyncFwd)
        local rtn = (Config.CurrentMode == "Normal" and Config.NormalRtn or Config.DesyncRtn)
        local dist = moveBypass(target, (phase <= 2 and fwd or rtn))
        if dist < 1.5 then
            if phase == 1 then if isLeft then autoLeftPhase = 2 else autoRightPhase = 2 end
            elseif phase == 2 then if isLeft then autoLeftPhase = 0 else autoRightPhase = 0 end
                task.delay(0.2, function() if isLeft then autoLeftPhase = 3 else autoRightPhase = 3 end end)
            elseif phase == 3 then if isLeft then autoLeftPhase = 4 else autoRightPhase = 4 end
            elseif phase == 4 then AutoLeftEnabled, AutoRightEnabled = false, false; forceStop(); updateUI() end
        end
    end)
    if isLeft then autoLeftConn = conn else autoRightConn = conn end
end

local function resolveRemote(path)
if not path or path == "" then return nil end
local obj = game
local cleaned = path:gsub("^game%.", "")
for segment in cleaned:gmatch("[^%.]+") do
if obj then obj = obj[segment] else return nil end
end
return obj
end

local function bomb(tableincrease, tries)
local maintable = {}
local spammedtable = {{}}
local z = spammedtable[1]
for i = 1, tableincrease do
local tableins = {}
table.insert(z, tableins)
z = tableins
end
local maximum = (499999 / (tableincrease + 2)) or 9999999
for i = 1, maximum do
table.insert(maintable, spammedtable)
if i % 5000 == 0 then task.wait() end
end
local remote = resolveRemote(CUSTOM_REMOTE_PATH)
if remote then
for i = 1, tries do
pcall(function()
if remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent") then
remote:FireServer(maintable)
elseif remote:IsA("RemoteFunction") then
remote:InvokeServer(maintable)
end
end)
end
end
end

local laggerEnabled = false
local laggerThread = nil

local function startLaggerLoop()
while laggerEnabled do
game:GetService("NetworkClient"):SetOutgoingKBPSLimit(math.huge)
task.spawn(function() bomb(LAGGER_CONFIG.TableIncrease, LAGGER_CONFIG.Tries) end)
task.wait(math.max(LAGGER_CONFIG.LoopWaitTime, 0.15))
end
end

local function stopLaggerLoop()
laggerEnabled = false
if laggerThread then coroutine.close(laggerThread); laggerThread = nil end
end

local function startLagger()
if laggerThread then return end
laggerEnabled = true
laggerThread = coroutine.create(startLaggerLoop)
coroutine.resume(laggerThread)
end

-- 🔥 ANTI-LAG
local function applyAntiLag(instance)
    if instance:IsA("ParticleEmitter") or instance:IsA("Trail") or instance:IsA("Smoke") or instance:IsA("Fire") then
        
        if optimized[instance] == nil then
            optimized[instance] = instance.Enabled
            instance.Enabled = false
        end

    elseif instance:IsA("Decal") or instance:IsA("Texture") then
        
        if optimized[instance] == nil then
            optimized[instance] = instance.Transparency
            instance.Transparency = 1
        end

    elseif instance:IsA("BasePart") then
        
        if not optimized[instance] then
            optimized[instance] = {
                instance.Material,
                instance.Reflectance,
                instance.CastShadow
            }
        end

        instance.Material = Enum.Material.Plastic
        instance.Reflectance = 0
        instance.CastShadow = false
    end
end

function enableOptimizer()
    if optimizerEnabled then return end
    optimizerEnabled = true

    savedLighting = {
        GlobalShadows = Lighting.GlobalShadows,
        FogStart = Lighting.FogStart,
        FogEnd = Lighting.FogEnd,
        Brightness = Lighting.Brightness,
        EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
        EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale
    }

    Lighting.GlobalShadows = false
    Lighting.FogStart = 0
    Lighting.FogEnd = 9e9
    Lighting.Brightness = 1
    Lighting.EnvironmentDiffuseScale = 0
    Lighting.EnvironmentSpecularScale = 0

    -- Quitar efectos (igual que tu script)
    for _, child in pairs(Lighting:GetChildren()) do
        if child:IsA("BloomEffect") or child:IsA("BlurEffect") or child:IsA("SunRaysEffect") or child:IsA("ColorCorrectionEffect") then
            if optimized[child] == nil then
                optimized[child] = child.Enabled
                child.Enabled = false
            end
        end
    end

    -- Aplicar a TODO + lógica ULTRA (CastShadow forzado)
    for _, v in ipairs(workspace:GetDescendants()) do
        applyAntiLag(v)

        if v:IsA("BasePart") then
            v.CastShadow = false
        end
    end

    -- Tiempo real
    optimizerConnection = workspace.DescendantAdded:Connect(function(desc)
        if optimizerEnabled then
            applyAntiLag(desc)

            if desc:IsA("BasePart") then
                desc.CastShadow = false
            end
        end
    end)
end

-- ❌ DESACTIVAR
function disableOptimizer()
    if not optimizerEnabled then return end
    optimizerEnabled = false

    if optimizerConnection then
        optimizerConnection:Disconnect()
        optimizerConnection = nil
    end

    for k,v in pairs(savedLighting) do
        Lighting[k] = v
    end

    for obj,val in pairs(optimized) do
        if obj and obj.Parent then
            if typeof(val) == "table" then
                obj.Material = val[1]
                obj.Reflectance = val[2]
                obj.CastShadow = val[3]

            elseif typeof(val) == "boolean" then
                pcall(function() obj.Enabled = val end)

            elseif typeof(val) == "number" then
                pcall(function() obj.Transparency = val end)
            end
        end
    end

    optimized = {}
end

local function applyXray(plot)
    if baseOriginalTransparencies[plot] then return end

    baseOriginalTransparencies[plot] = {}

    local oldHL = plot:FindFirstChild("BaseXRayHighlight")
    if oldHL then
        oldHL:Destroy()
    end

    for _, part in ipairs(plot:GetDescendants()) do

            if oldPartHL then
                oldPartHL:Destroy()

            if part.Transparency < 0.6 then

        if desc:IsA("BasePart") then

        end
    end)

-- Ocultar accesorio específico
local function removeAccessory(accessory)
    if not ANTI_FPS_DEVOURER.hiddenAccessories[accessory] then

        accessory.Parent = nil

local function scanModel(model)
    for _, obj in ipairs(model:GetDescendants()) do
if obj:IsA("Accessory")
or obj:IsA("Shirt")
or obj:IsA("Pants")
or obj:IsA("ShirtGraphic") then
    removeAccessory(obj)
end
    end
end

function enableAntiFPSDevourer()
    if ANTI_FPS_DEVOURER.enabled then return end
    ANTI_FPS_DEVOURER.enabled = true

    -- Escanea cualquier accesorio o ropa 3d de modelos y usuarios 
    for _, obj in ipairs(workspace:GetDescendants()) do
if obj:IsA("Accessory")
or obj:IsA("Shirt")
or obj:IsA("Pants")
or obj:IsA("ShirtGraphic") then
    removeAccessory(obj)
end
    end

    -- Detectar nuevos objetos 
    local conn = workspace.DescendantAdded:Connect(function(obj)
        if ANTI_FPS_DEVOURER.enabled then
if obj:IsA("Accessory")
or obj:IsA("Shirt")
or obj:IsA("Pants")
or obj:IsA("ShirtGraphic") then
    removeAccessory(obj)
end
        end
    end)

    table.insert(ANTI_FPS_DEVOURER.connections, conn)
end

function disableAntiFPSDevourer()
    if not ANTI_FPS_DEVOURER.enabled then return end
    ANTI_FPS_DEVOURER.enabled = false

    -- Desconectar
    for _, conn in ipairs(ANTI_FPS_DEVOURER.connections) do
        if typeof(conn) == "RBXScriptConnection" then
            conn:Disconnect()
        end
    end

    ANTI_FPS_DEVOURER.connections = {}

    -- Restaurar accesorios
for accessory, originalParent in pairs(ANTI_FPS_DEVOURER.hiddenAccessories) do
    if accessory and accessory.Parent ~= originalParent then
        if originalParent then
            accessory.Parent = originalParent
        end
    end
end

    ANTI_FPS_DEVOURER.hiddenAccessories = {}
end

local MELEE_RANGE = 45
local MELEE_ONLY_ENEMIES = true

local meleeEnabled = false

local function isValidMeleeTarget(humanoid, rootPart)
    if not (humanoid and rootPart) then return false end
    if humanoid.Health <= 0 then return false end
    
    if MELEE_ONLY_ENEMIES then
        local targetPlayer = Players:GetPlayerFromCharacter(humanoid.Parent)
        if not targetPlayer or targetPlayer == player then
            return false
        end
    end
    
    return true
end

local function getClosestMeleeTarget(hrp)
    local closest
    local minDist = MELEE_RANGE
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p == player then continue end
        local char = p.Character
        if not char then continue end
        
        local targetHrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        
        if isValidMeleeTarget(hum, targetHrp) then
            local dist = (targetHrp.Position - hrp.Position).Magnitude
            if dist < minDist then
                minDist = dist
                closest = targetHrp
            end
        end
    end
    
    return closest
end

local function createMeleeAimbot(char)
    local hrp = char:WaitForChild("HumanoidRootPart", 8)
    local humanoid = char:WaitForChild("Humanoid", 8)
    
    if not (hrp and humanoid) then return end
    
    if meleeConnection then
        meleeConnection:Disconnect()
    end
    
    meleeConnection = RunService.RenderStepped:Connect(function()
        if not char.Parent or not meleeEnabled then return end
        
        local target = getClosestMeleeTarget(hrp)
        
        if target then
            humanoid.AutoRotate = false
            -- Rotación directa sin usar objetos de orientación
            local targetPos = Vector3.new(target.Position.X, hrp.Position.Y, target.Position.Z)
            hrp.CFrame = CFrame.new(hrp.Position, targetPos)
        else
            humanoid.AutoRotate = true
        end
    end)
end

    meleeEnabled = false
    
    if meleeConnection then
        meleeConnection:Disconnect()
        meleeConnection = nil
    end
    
    if character and character:FindFirstChild("Humanoid") then
        character.Humanoid.AutoRotate = true
    end
end

local function pathMove(point)

local char = player.Character
if not char then return end

local humanoid = char:FindFirstChild("Humanoid")
local root = char:FindFirstChild("HumanoidRootPart")

if not humanoid or not root then return end

local path = PathfindingService:CreatePath({
AgentRadius = 2,
AgentHeight = 5,
AgentCanJump = false
})

path:ComputeAsync(root.Position, point)

local waypoints = {}

if path.Status == Enum.PathStatus.Success then
waypoints = path:GetWaypoints()
else
table.insert(waypoints,{Position = point})
end

local index = 2

while routeRunning and index <= #waypoints do

if not routeRunning then
root.AssemblyLinearVelocity = Vector3.new(0,root.AssemblyLinearVelocity.Y,0)
break
end

local target = waypoints[index].Position

local direction = target - root.Position
direction = Vector3.new(direction.X,0,direction.Z)

local distance = direction.Magnitude

if distance < 2 then
index += 1
else

direction = direction.Unit

local currentSpeed

if stealMode then
currentSpeed = getStealSpeed()
else
currentSpeed = getSpeed()
end

root.AssemblyLinearVelocity = Vector3.new(
direction.X * currentSpeed,
root.AssemblyLinearVelocity.Y,
direction.Z * currentSpeed
)

end

task.wait()

end

root.AssemblyLinearVelocity = Vector3.new(0,root.AssemblyLinearVelocity.Y,0)

end

local function directMove(point,myRouteId)

local char = player.Character
if not char then return end

local root = char:FindFirstChild("HumanoidRootPart")
if not root then return end

while routeRunning and currentRouteId == myRouteId do

local direction = point - root.Position
direction = Vector3.new(direction.X,0,direction.Z)

if direction.Magnitude < 2 then
break
end

direction = direction.Unit

local speed

if stealMode then
speed = getStealSpeed()
else
speed = getSpeed()
end

root.AssemblyLinearVelocity = Vector3.new(
direction.X * speed,
root.AssemblyLinearVelocity.Y,
direction.Z * speed
)

task.wait()

end

root.AssemblyLinearVelocity = Vector3.new(0,root.AssemblyLinearVelocity.Y,0)

end

local function walkRoute(route)

    local myRouteId = currentRouteId
    local activeRoute = route

    routeRunning = true
    stealMode = false

    -- ======================
    -- IDA
    -- ======================

    -- Start → Route1 (PATHFINDING)
    if myRouteId ~= currentRouteId then return end
    pathMove(route.Route[1])

    -- Route1 → Route2
    if myRouteId ~= currentRouteId then return end
    directMove(route.Route[2], myRouteId)

    -- Route2 → Route3
    if myRouteId ~= currentRouteId then return end
    directMove(route.Route[3], myRouteId)

    -- ======================
    -- STEAL
    -- ======================

    stealMode = true

    local cd = tonumber(cooldownBox.Text)
    if cd and cd > 0 then
        task.wait(cd)
    end

    -- ======================
    -- REGRESO
    -- ======================

    -- Route3 → Route2
    if myRouteId ~= currentRouteId then return end
    directMove(route.Route[2], myRouteId)

    -- Route2 → Route1
    if myRouteId ~= currentRouteId then return end
    directMove(route.Route[1], myRouteId)

    -- 🔥 Route1 → antiguo StartPoint (Route[4])
    if myRouteId ~= currentRouteId then return end
    if route.Route[4] then
        directMove(route.Route[4], myRouteId)
    end

    -- 🔥 antiguo StartPoint → nuevo StartPoint (Point 5)
    if myRouteId ~= currentRouteId then return end
    directMove(activeRoute.StartPoint, myRouteId)

    -- ======================
    -- TERMINAR
    -- ======================

    clearAutoWalkParts()

    if activeRoute == routes.START_A then
        leftEnabled = false
        leftButton.BackgroundColor3 = Color3.fromRGB(25,25,25)
    end

    if activeRoute == routes.START_B then
        rightEnabled = false
        rightButton.BackgroundColor3 = Color3.fromRGB(25,25,25)
    end

    selectedRoute = nil
    routeRunning = false

    resetPlayButton()

end

local function startAntiRagdoll()
    if Connections.antiRagdoll then return end
    
    Connections.antiRagdoll = game:GetService("RunService").Heartbeat:Connect(function()
        if not AntiRagdollEnabled then return end
        
        local char = player.Character
        if not char then return end
        
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not (hum and root) then return end
        
        local s = hum:GetState()
        local ragdolled = (s == Enum.HumanoidStateType.Physics
        or s == Enum.HumanoidStateType.Ragdoll
        or s == Enum.HumanoidStateType.FallingDown)
        
        local endTime = player:GetAttribute("RagdollEndTime")
        if endTime and (endTime - workspace:GetServerTimeNow()) > 0 then 
            ragdolled = true 
        end
        
        if ragdolled then
            pcall(function() player:SetAttribute("RagdollEndTime", workspace:GetServerTimeNow()) end)
            
            for _, d in ipairs(char:GetDescendants()) do
                if d:IsA("BallSocketConstraint") or (d:IsA("Attachment") and d.Name:find("RagdollAttachment")) then 
                    d:Destroy() 
                end
            end
            
            for _, obj in ipairs(char:GetDescendants()) do
                if obj:IsA("Motor6D") and obj.Enabled == false then 
                    obj.Enabled = true 
                end
            end
            
            if hum.Health > 0 then 
                hum:ChangeState(Enum.HumanoidStateType.Running) 
            end
            
            workspace.CurrentCamera.CameraSubject = hum
            root.Anchored = false
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

local function stopAntiRagdoll()
    if Connections.antiRagdoll then 
        Connections.antiRagdoll:Disconnect() 
        Connections.antiRagdoll = nil 
    end
end

        if enabled then
            if animate then
                savedAnimate = animate
                animate.Disabled = true
            end

            if humanoid then
                local animator = humanoid:FindFirstChildOfClass("Animator")

                if animator then
                    for _,track in ipairs(animator:GetPlayingAnimationTracks()) do
                        track:Stop(0)
                        track:Destroy()
                    end
                end

                humanoid:ChangeState(Enum.HumanoidStateType.Physics)
                humanoid:ChangeState(Enum.HumanoidStateType.Running)
            end

        else
            if savedAnimate then
                savedAnimate.Disabled = false
                savedAnimate = nil
            end
        end


    elseif text == "Instant Grab" then
        autoStealEnabled = enabled

        if enabled then

            task.spawn(function()

                repeat task.wait() until progressBarBg and progressFill and percentLabel

                progressBarBg.Visible = true

                if instantGrabConnection then
                    instantGrabConnection:Disconnect()
                end

                while autoStealEnabled do

                    local char = lp.Character
                    if char then

                        local root = char:FindFirstChild("HumanoidRootPart")
                        if root then

                            if fastStealEnabled and autoStealEnabled then
                                grabRadius = 10
                            end

                            local prompt = findNearestSteal(root)

                            if prompt then
                                progressBarBg.Visible = true
                                local start = tick()

                                while autoStealEnabled and findNearestSteal(root) == prompt do

                                    local duration = (fastStealEnabled and autoStealEnabled) and 0.10 or 0.10
                                    local p = math.clamp((tick() - start) / duration, 0, 1)

                                    progressFill.Size = UDim2.new(p,0,1,0)

                                        pcall(fireproximityprompt, prompt)

                                        pcall(fireproximityprompt, prompt)
                                        start = tick()
                                    end

                                    task.wait()
                                end
                            end

                    task.wait(0.15)

local debounce = false

local wfConns = {}
local wfActive = false

local function startWalkFling()

    wfActive = true

    table.insert(wfConns, RunService.Stepped:Connect(function()

        if not wfActive then return end

        for _,p in ipairs(Players:GetPlayers()) do
            if p ~= player and p.Character then

                for _,part in ipairs(p.Character:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end

            end
        end

    end))

    local co = coroutine.create(function()

        while wfActive do

            RunService.Heartbeat:Wait()

            local char = player.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")

            if not root then
                RunService.Heartbeat:Wait()
                continue
            end

            local vel = root.Velocity

            root.Velocity = vel * 10000 + Vector3.new(0,10000,0)

            RunService.RenderStepped:Wait()

            if root and root.Parent then
                root.Velocity = vel
            end

            RunService.Stepped:Wait()

            if root and root.Parent then
                root.Velocity = vel + Vector3.new(0,0.1,0)
            end

        end

    end)

    coroutine.resume(co)
    table.insert(wfConns, co)

end


local function stopWalkFling()

    wfActive = false

    for _,c in ipairs(wfConns) do
        if typeof(c) == "RBXScriptConnection" then
            c:Disconnect()
        elseif typeof(c) == "thread" then
            pcall(task.cancel, c)
        end
    end

    wfConns = {}

end


local function doDrop()

    local char = player.Character
    if not char then return end

    local humanoid = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")

    if not humanoid or not root then return end

    -- guardar altura original
    local originalY = root.Position.Y

    -- registrar salto
    humanoid:ChangeState(Enum.HumanoidStateType.Jumping)

    -- fuerza real del salto
    root.AssemblyLinearVelocity = Vector3.new(
        root.AssemblyLinearVelocity.X,
        100,
        root.AssemblyLinearVelocity.Z
    )

    -- bloquear solo la altura (no movimiento)
    local start = tick()
    while tick() - start < 0.2 do

        local pos = root.Position
        root.CFrame = CFrame.new(pos.X, originalY, pos.Z) * CFrame.Angles(0, root.Orientation.Y * math.pi/180, 0)

        task.wait()

    end

end

local function makeDraggable(obj)
    local dragging, dragInput, dragStart, startPos
    obj.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = obj.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if obj.Name == "ToggleButton" then savePos(obj.Position) end
                end
            end)
        end
    end)
    obj.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            local newPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            
            local screenSize = sg.AbsoluteSize
            local objSize = obj.AbsoluteSize
            local x = math.clamp(newPos.X.Scale * screenSize.X + newPos.X.Offset, 0, screenSize.X - objSize.X)
            local y = math.clamp(newPos.Y.Scale * screenSize.Y + newPos.Y.Offset, 0, screenSize.Y - objSize.Y)
            
            obj.Position = UDim2.new(0, x, 0, y)
        end
    end)
end

local function getNearest()
	local char = lp.Character
	if not char then return nil end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return nil end

	local nearest = nil
	local nearestDist = LOCK_RADIUS

	for _,plr in ipairs(Players:GetPlayers()) do
		if plr ~= lp then
			local pc = plr.Character
			local phrp = pc and pc:FindFirstChild("HumanoidRootPart")
			if phrp then
				local d = (phrp.Position - hrp.Position).Magnitude
				if d <= nearestDist then
					nearest = plr
					nearestDist = d
				end
			end
		end
	end

	return nearest
end

local function getBat()
	for _,t in ipairs(lp.Backpack:GetChildren()) do
		if t:IsA("Tool") and string.find(string.lower(t.Name),"bat",1,true) then
			return t
		end
	end

	local char = lp.Character
	if char then
		for _,t in ipairs(char:GetChildren()) do
			if t:IsA("Tool") and string.find(string.lower(t.Name),"bat",1,true) then
				return t
			end
		end
	end

	return nil
end

local function startLock()

	local char = lp.Character
	if not char then return end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	lockAtt = Instance.new("Attachment", hrp)

	lockLv = Instance.new("LinearVelocity", hrp)
	lockLv.Attachment0 = lockAtt
	lockLv.MaxForce = 50000
	lockLv.RelativeTo = Enum.ActuatorRelativeTo.World
	lockLv.Enabled = false

	lockHbConn = RunService.Heartbeat:Connect(function()

		if not currentTarget then
			currentTarget = getNearest()
		end

		local targetPlayer = currentTarget
		if not targetPlayer then
			lockLv.Enabled = false
			return
		end

		local tChar = targetPlayer.Character
		local tHrp = tChar and tChar:FindFirstChild("HumanoidRootPart")

		if not tHrp then
			currentTarget = nil
			lockLv.Enabled = false
			return
		end

		lockLv.Enabled = true

		-- 🔥 DIRECTO AL CUERPO (SIN PREDICCIÓN)
		local targetPos = tHrp.Position

		local dir = targetPos - hrp.Position
		local distance = dir.Magnitude

		local stopDistance = 1.2

		if distance > stopDistance then
			-- ir directo al target
			lockLv.VectorVelocity = dir.Unit * LOCK_SPEED
		else
			-- 🧲 pegarse (copiar velocidad)
			lockLv.VectorVelocity = tHrp.AssemblyLinearVelocity * 1.05
		end

	end)

end


local function stopLock()

	currentTarget = nil

	if lockHbConn then
		lockHbConn:Disconnect()
		lockHbConn = nil
	end

	if lockLv then
		lockLv:Destroy()
		lockLv = nil
	end

	if lockAtt then
		lockAtt:Destroy()
		lockAtt = nil
	end

end

local medusaPart

local MedusaInitialized = false

local function InitMedusa()

    if MedusaInitialized then return end
    MedusaInitialized = true

    local function createRadius()
        if medusaPart then medusaPart:Destroy() end

        medusaPart = Instance.new("Part")
        medusaPart.Name = "MedusaRadius"
        medusaPart.Anchored = true
        medusaPart.CanCollide = false
        medusaPart.Transparency = 0
        medusaPart.Material = Enum.Material.Neon
        medusaPart.Color = Color3.fromRGB(255, 40, 40)
        medusaPart.Shape = Enum.PartType.Cylinder
        medusaPart.Size = Vector3.new(0.2, MEDUSA_RADIUS*2, MEDUSA_RADIUS*2)
        medusaPart.Parent = workspace
    end

    local function isMedusaEquipped()
        local char = lp.Character
        if not char then return nil end

        for _, tool in ipairs(char:GetChildren()) do
            if tool:IsA("Tool") and tool.Name == "Medusa's Head" then
                return tool
            end
        end
        return nil
    end

    createRadius()

    -- VISUAL
    RunService.RenderStepped:Connect(function()

        if not AutoMedusaEnabled then
            if medusaPart then medusaPart.Transparency = 1 end
            return
        end

        medusaPart.Transparency = 0

        local char = lp.Character
        if not char then return end

        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end

        medusaPart.CFrame =
            CFrame.new(root.Position + Vector3.new(0, -2.5, 0))
            * CFrame.Angles(0, 0, math.rad(90))
    end)

    -- SPAM
    RunService.Heartbeat:Connect(function()

        if not AutoMedusaEnabled then return end

        local char = lp.Character
        if not char then return end

        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end

        local tool = isMedusaEquipped()
        if not tool then return end

        if tick() - lastUse < SPAM_DELAY then return end

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= lp then
                local pChar = plr.Character
                local pRoot = pChar and pChar:FindFirstChild("HumanoidRootPart")
                if pRoot then
                    if (pRoot.Position - root.Position).Magnitude <= MEDUSA_RADIUS then
                        tool:Activate()
                        lastUse = tick()
                        break
                    end
                end
            end
        end
    end)
end

function getGroundPosition()
    local char = game.Players.LocalPlayer.Character
    if not char then return end

    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local rayOrigin = root.Position
    local rayDirection = Vector3.new(0, -500, 0)

    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {char}
    params.FilterType = Enum.RaycastFilterType.Blacklist

    local result = workspace:Raycast(rayOrigin, rayDirection, params)

    if result then
        return result.Position
    end
end

local function isMyBase(pn) local pl=workspace:FindFirstChild("Plots"); if not pl then return false end; local p=pl:FindFirstChild(pn); if not p then return false end; local s=p:FindFirstChild("PlotSign"); if not s then return false end; local yb=s:FindFirstChild("YourBase"); return yb and yb:IsA("BillboardGui") and yb.Enabled end
local function scanPlot(plot) if not plot or not plot:IsA("Model") then return end; if isMyBase(plot.Name) then return end; local pods=plot:FindFirstChild("AnimalPodiums"); if not pods then return end; for _,pod in ipairs(pods:GetChildren()) do if pod:IsA("Model") and pod:FindFirstChild("Base") then table.insert(animalCache,{plot=plot.Name,slot=pod.Name,worldPosition=pod:GetPivot().Position,uid=plot.Name.."_"..pod.Name}) end end end
local function findPromptForAnimal(ad) if not ad then return nil end; local cp=promptCache[ad.uid]; if cp and cp.Parent then return cp end; local plots=workspace:FindFirstChild("Plots"); local plot=plots and plots:FindFirstChild(ad.plot); local pods=plot and plot:FindFirstChild("AnimalPodiums"); local pod=pods and pods:FindFirstChild(ad.slot); local base=pod and pod:FindFirstChild("Base"); local sp=base and base:FindFirstChild("Spawn"); local att=sp and sp:FindFirstChild("PromptAttachment"); if att then for _,p in ipairs(att:GetChildren()) do if p:IsA("ProximityPrompt") then promptCache[ad.uid]=p; return p end end end return nil end
local function buildCallbacks(prompt) if stealCache[prompt] then return end; local data={hold={},trigger={},ready=true}; pcall(function() if getconnections then for _,c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do if type(c.Function)=="function" then table.insert(data.hold,c.Function) end end; for _,c in ipairs(getconnections(prompt.Triggered)) do if type(c.Function)=="function" then table.insert(data.trigger,c.Function) end end end end); if #data.hold>0 or #data.trigger>0 then stealCache[prompt]=data end end

local function execSteal(prompt)
    local data=stealCache[prompt]; if not data or not data.ready then return false end; data.ready=false; isStealing=true; stealStartTime=tick()
    if ProgressLabel then ProgressLabel.Text="STEALING..." end
    if progressConn then progressConn:Disconnect() end
    progressConn=RunService.Heartbeat:Connect(function() if not isStealing then progressConn:Disconnect(); return end; local prog=math.clamp((tick()-stealStartTime)/STEAL_DURATION,0,1); if ProgressBarFill then ProgressBarFill.Size=UDim2.new(prog,0,1,0) end; if ProgressPctLabel then ProgressPctLabel.Text=math.floor(prog*100).."%" end end)
    task.spawn(function() for _,f in ipairs(data.hold) do task.spawn(f) end; task.wait(STEAL_DURATION); for _,f in ipairs(data.trigger) do task.spawn(f) end; if progressConn then progressConn:Disconnect() end; if ProgressLabel then ProgressLabel.Text="READY" end; if ProgressPctLabel then ProgressPctLabel.Text="" end; if ProgressBarFill then ProgressBarFill.Size=UDim2.new(0,0,1,0) end; data.ready=true; isStealing=false end)
    return true
end

local function nearestAnimal() local h=getHRP(); if not h then return nil end; local best,bestD=nil,math.huge; for _,ad in ipairs(animalCache) do if not isMyBase(ad.plot) and ad.worldPosition then local d=(h.Position-ad.worldPosition).Magnitude; if d<bestD then bestD=d; best=ad end end end; return best end

function startAutoSteal() if autoStealConn then return end; autoStealConn=RunService.Heartbeat:Connect(function() if not autoStealEnabled or isStealing then return end; local target=nearestAnimal(); if not target then return end; local h=getHRP(); if not h then return end; if (h.Position-target.worldPosition).Magnitude>STEAL_RADIUS then return end; local prompt=promptCache[target.uid]; if not prompt or not prompt.Parent then prompt=findPromptForAnimal(target) end; if prompt then buildCallbacks(prompt); execSteal(prompt) end end) end
function stopAutoSteal() if autoStealConn then autoStealConn:Disconnect(); autoStealConn=nil end; isStealing=false; if progressConn then progressConn:Disconnect() end; if ProgressBarFill then ProgressBarFill.Size=UDim2.new(0,0,1,0) end; if ProgressLabel then ProgressLabel.Text="READY" end; if ProgressPctLabel then ProgressPctLabel.Text="" end end

-- Inicio de Escaneo y Ejecución
task.spawn(function()
    task.wait(2)
    local plots=workspace:WaitForChild("Plots",10)
    if plots then
        for _,p in ipairs(plots:GetChildren()) do if p:IsA("Model") then scanPlot(p) end end
        plots.ChildAdded:Connect(function(p) if p:IsA("Model") then task.wait(0.5); scanPlot(p) end end)
        startAutoSteal()
    end
end)
end)

task.spawn(function()
-- STEAL CUSTOMIZER (PRO - PRESETS FIXED)

local HttpService = game:GetService("HttpService")
local SpeedCustomizer = {
    Enabled = false,
    SpeedUI = nil,
    StealValue = 20,
    P1 = 12,
    P2 = 17,
    P3 = 20,
    HeartbeatConn = nil,
    character = nil,
    hrp = nil,
    hum = nil,
    FileName = "Kawatan_StealPresets.json"
}

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local lp = Players.LocalPlayer

-- ELIMINAR DUPLICADOS SI YA EXISTE
local oldUI = game:GetService("CoreGui"):FindFirstChild("Kawatan_SpeedUI")
if oldUI then oldUI:Destroy() end

-- GUARDADO EN DISCO
local function SaveConfigToDisk()
    local data = {
        Enabled = SpeedCustomizer.Enabled,
        Steal = SpeedCustomizer.StealValue,
        P1 = SpeedCustomizer.P1,
        P2 = SpeedCustomizer.P2,
        P3 = SpeedCustomizer.P3,
        Pos = {
            X = SpeedCustomizer.SpeedUI.SpeedMainFrame.Position.X.Scale,
            XO = SpeedCustomizer.SpeedUI.SpeedMainFrame.Position.X.Offset,
            Y = SpeedCustomizer.SpeedUI.SpeedMainFrame.Position.Y.Scale,
            YO = SpeedCustomizer.SpeedUI.SpeedMainFrame.Position.Y.Offset
        }
    }
    if writefile then writefile(SpeedCustomizer.FileName, HttpService:JSONEncode(data)) end
end

local loadedData = {}
if isfile and isfile(SpeedCustomizer.FileName) then
    pcall(function() loadedData = HttpService:JSONDecode(readfile(SpeedCustomizer.FileName)) end)
end

SpeedCustomizer.Enabled = loadedData.Enabled or false
SpeedCustomizer.StealValue = math.clamp(loadedData.Steal or 20, 1, 20)
SpeedCustomizer.P1 = math.clamp(loadedData.P1 or 12, 1, 20)
SpeedCustomizer.P2 = math.clamp(loadedData.P2 or 17, 1, 20)
SpeedCustomizer.P3 = math.clamp(loadedData.P3 or 20, 1, 20)

local function setupChar(char)
    SpeedCustomizer.character = char
    SpeedCustomizer.hrp = char:WaitForChild("HumanoidRootPart")
    SpeedCustomizer.hum = char:WaitForChild("Humanoid")
end

local function createUI()
    local sg = Instance.new("ScreenGui", game:GetService("CoreGui"))
    sg.Name = "Kawatan_SpeedUI"

    local mf = Instance.new("Frame", sg)
    mf.Name = "SpeedMainFrame"
    mf.Size = UDim2.new(0, 200, 0, 35) -- Cerrado por defecto
    mf.Position = loadedData.Pos and UDim2.new(loadedData.Pos.X, loadedData.Pos.XO, loadedData.Pos.Y, loadedData.Pos.YO) or UDim2.new(0.5, -100, 0.4, 0)
    mf.BackgroundColor3 = Color3.fromRGB(12, 12, 15)
    mf.Active = true
    mf.Draggable = true
    Instance.new("UICorner", mf)
    Instance.new("UIStroke", mf).Color = Color3.fromRGB(50, 150, 250)

    local CollapseBtn = Instance.new("TextButton", mf)
    CollapseBtn.Size = UDim2.new(0, 30, 0, 30); CollapseBtn.Position = UDim2.new(1, -35, 0, 2)
    CollapseBtn.Text = "v"; CollapseBtn.TextColor3 = Color3.new(1, 1, 1); CollapseBtn.BackgroundTransparency = 1; CollapseBtn.TextSize = 20; CollapseBtn.Font = Enum.Font.GothamBold

    local Title = Instance.new("TextLabel", mf)
    Title.Size = UDim2.new(1, -45, 0, 30); Title.Position = UDim2.new(0, 10, 0, 2)
    Title.Text = "Steal Customizer"; Title.TextColor3 = Color3.fromRGB(80, 180, 255); Title.BackgroundTransparency = 1; Title.TextXAlignment = Enum.TextXAlignment.Left; Title.Font = Enum.Font.GothamBold

    local toggle = Instance.new("TextButton", mf)
    toggle.Size = UDim2.new(1, -20, 0, 25); toggle.Position = UDim2.new(0, 10, 0, 40)
    toggle.BackgroundColor3 = SpeedCustomizer.Enabled and Color3.fromRGB(0, 170, 0) or Color3.fromRGB(170, 0, 0)
    toggle.Text = SpeedCustomizer.Enabled and "ACTIVADO" or "DESACTIVADO"
    toggle.TextColor3 = Color3.new(1, 1, 1); toggle.Font = Enum.Font.GothamBold; toggle.Visible = false; Instance.new("UICorner", toggle)

    local tInp = Instance.new("TextBox", mf)
    tInp.Size = UDim2.new(0, 65, 0, 20); tInp.Position = UDim2.new(1, -75, 0, 75)
    tInp.BackgroundColor3 = Color3.fromRGB(25, 25, 30); tInp.Text = tostring(SpeedCustomizer.StealValue); tInp.TextColor3 = Color3.new(1, 1, 1); tInp.Font = Enum.Font.GothamBold; tInp.ClearTextOnFocus = false; tInp.Visible = false; Instance.new("UICorner", tInp)

    local lab = Instance.new("TextLabel", mf)
    lab.Size = UDim2.new(0.5, 0, 0, 20); lab.Position = UDim2.new(0, 10, 0, 75); lab.Text = "Steal Speed"; lab.TextColor3 = Color3.fromRGB(200, 200, 200); lab.BackgroundTransparency = 1; lab.Visible = false; lab.TextXAlignment = Enum.TextXAlignment.Left

    -- BOTONES (NORMAL + PRESETS)
    local presetFrame = Instance.new("Frame", mf)
    presetFrame.Size = UDim2.new(1, -20, 0, 30); presetFrame.Position = UDim2.new(0, 10, 1, 5); presetFrame.BackgroundTransparency = 1

    local buttons = {}

    local function updateButtonColors()
        for key, btn in pairs(buttons) do
            if key == "Normal" then
                btn.BackgroundColor3 = (not SpeedCustomizer.Enabled) and Color3.fromRGB(130, 0, 255) or Color3.fromRGB(25, 25, 30)
            else
                btn.BackgroundColor3 = (SpeedCustomizer.Enabled and SpeedCustomizer.StealValue == SpeedCustomizer[key]) and Color3.fromRGB(130, 0, 255) or Color3.fromRGB(25, 25, 30)
            end
        end
    end

    local function createPresetBtn(val, xPos, key, isNormal)
        local btn = Instance.new("TextButton", presetFrame)
        btn.Size = UDim2.new(0.22, 0, 1, 0); btn.Position = UDim2.new(xPos, 0, 0, 0)
        btn.BackgroundColor3 = Color3.fromRGB(25, 25, 30); btn.Text = tostring(val); btn.TextColor3 = Color3.new(1, 1, 1); btn.Font = Enum.Font.GothamBold
        Instance.new("UICorner", btn); Instance.new("UIStroke", btn).Color = Color3.fromRGB(50, 150, 250)
        
        buttons[key] = btn

        btn.MouseButton1Click:Connect(function()
            if isNormal then
                SpeedCustomizer.Enabled = false
                toggle.Text = "DESACTIVADO"
                toggle.BackgroundColor3 = Color3.fromRGB(170, 0, 0)
            else
                SpeedCustomizer.Enabled = true
                toggle.Text = "ACTIVADO"
                toggle.BackgroundColor3 = Color3.fromRGB(0, 170, 0)
                SpeedCustomizer.StealValue = SpeedCustomizer[key]
                tInp.Text = tostring(SpeedCustomizer.StealValue)
            end
            updateButtonColors()
            SaveConfigToDisk()
        end)
        return btn
    end

    local normBtn = createPresetBtn("Normal", 0, "Normal", true)
    local p1Btn = createPresetBtn(SpeedCustomizer.P1, 0.26, "P1", false)
    local p2Btn = createPresetBtn(SpeedCustomizer.P2, 0.52, "P2", false)
    local p3Btn = createPresetBtn(SpeedCustomizer.P3, 0.78, "P3", false)

    -- INPUTS PARA EDITAR
    local function createEditBox(txt, key, y, btnRef)
        local box = Instance.new("TextBox", mf)
        box.Size = UDim2.new(0, 65, 0, 20); box.Position = UDim2.new(1, -75, 0, y)
        box.BackgroundColor3 = Color3.fromRGB(20, 20, 25); box.Text = tostring(SpeedCustomizer[key])
        box.TextColor3 = Color3.new(1, 1, 1); box.Font = Enum.Font.GothamBold; box.ClearTextOnFocus = false; box.Visible = false; Instance.new("UICorner", box)
        
        local label = Instance.new("TextLabel", mf)
        label.Size = UDim2.new(0.5, 0, 0, 20); label.Position = UDim2.new(0, 10, 0, y)
        label.Text = txt; label.TextColor3 = Color3.fromRGB(150, 150, 150); label.BackgroundTransparency = 1; label.Visible = false; label.TextXAlignment = Enum.TextXAlignment.Left

        box.FocusLost:Connect(function()
            local n = math.clamp(tonumber(box.Text) or SpeedCustomizer[key], 1, 20)
            SpeedCustomizer[key] = n
            box.Text = tostring(n)
            btnRef.Text = tostring(n)
            updateButtonColors()
            SaveConfigToDisk()
        end)
        return box, label
    end

    local e1, l1 = createEditBox("Set Preset 1", "P1", 105, p1Btn)
    local e2, l2 = createEditBox("Set Preset 2", "P2", 130, p2Btn)
    local e3, l3 = createEditBox("Set Preset 3", "P3", 155, p3Btn)

    local open = false
    CollapseBtn.MouseButton1Click:Connect(function()
        open = not open
        toggle.Visible = open; tInp.Visible = open; lab.Visible = open
        e1.Visible = open; l1.Visible = open; e2.Visible = open; l2.Visible = open; e3.Visible = open; l3.Visible = open
        mf.Size = open and UDim2.new(0, 200, 0, 185) or UDim2.new(0, 200, 0, 35)
        CollapseBtn.Text = open and "^" or "v"
    end)

    tInp.FocusLost:Connect(function() 
        SpeedCustomizer.StealValue = math.clamp(tonumber(tInp.Text) or SpeedCustomizer.StealValue, 1, 20)
        tInp.Text = tostring(SpeedCustomizer.StealValue)
        updateButtonColors()
        SaveConfigToDisk()
    end)

    toggle.MouseButton1Click:Connect(function()
        SpeedCustomizer.Enabled = not SpeedCustomizer.Enabled
        toggle.Text = SpeedCustomizer.Enabled and "ACTIVADO" or "DESACTIVADO"
        toggle.BackgroundColor3 = SpeedCustomizer.Enabled and Color3.fromRGB(0, 170, 0) or Color3.fromRGB(170, 0, 0)
        updateButtonColors()
        SaveConfigToDisk()
    end)

    mf:GetPropertyChangedSignal("Position"):Connect(SaveConfigToDisk)
    SpeedCustomizer.SpeedUI = sg
    updateButtonColors()
end

local function enable()
    if not SpeedCustomizer.SpeedUI then createUI() end
    if lp.Character then setupChar(lp.Character) end
    lp.CharacterAdded:Connect(setupChar)

    RunService.Heartbeat:Connect(function()
        if not SpeedCustomizer.Enabled or not SpeedCustomizer.hrp or not SpeedCustomizer.hum then return end
        local moveDir = SpeedCustomizer.hum.MoveDirection
        if moveDir.Magnitude > 0 and SpeedCustomizer.hum.WalkSpeed > 0.1 then
            if SpeedCustomizer.hum.WalkSpeed < 34 then
                SpeedCustomizer.hrp.AssemblyLinearVelocity = Vector3.new(moveDir.X * SpeedCustomizer.StealValue, SpeedCustomizer.hrp.AssemblyLinearVelocity.Y, moveDir.Z * SpeedCustomizer.StealValue)
            end
        end
    end)
end

enable()
end)
